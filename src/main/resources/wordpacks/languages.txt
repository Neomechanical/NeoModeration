# Everyday words in one language that another language's list bans. They pass
# for everyone on a server whose locale is the language named, and for a player
# whose game language is, when the same message has another everyday word of
# that language (common-words.txt): "quel retard ce train" passes for a French
# player, "shut up retard" does not. "<language>: <word> <word> ..."; several
# languages may share a line, and a language such as "pt_pt" names one
# country's form, "sv" every Swedish one.
#
# French and Catalan: "en retard", late
fr: retard retards
ca: retard retards
# Spanish: Kike, short for Enrique; "una porra", a betting pool
es: kike porra
# Portuguese in Portugal: "os putos", the kids; "a bicha", the queue
pt_pt: puto putos bicha bichas
# Swedish and Danish: "slut", the end; Swedish "fick", got
sv: slut fick
da: slut
# Danish and Norwegian: "fag", a school subject
da no nb nn: fag
# Croatian, Serbian and Bosnian: "dva puta", twice
hr sr bs: puta
# Italian: "verga", a rod, and the writer Giovanni Verga
it: verga
